//1.
#include<stdio.h>
#include<stdlib.h>
#include<unistd.h>
#include<pthread.h>

//A normal C function that is executed as a thread
//when its name is specified in pthread_create()
void *helloworld(void *vargp){
	
	sleep(1);
	printf("Hello World\n");
	return NULL;
	
}

int main(){
	pthread_t thread_id;
	printf("Before Thread\n");
	pthread_create(&thread_id, NULL, helloworld, NULL);
	pthread_join(thread_id, NULL);
	printf("After Thread\n");
	exit(0);
}

Output:

[2021ict37@fedora ~]$ ./thread1
Before Thread
Hello World
After Thread

//2.
#include<stdio.h>
#include<stdlib.h>
#include<unistd.h>
#include<pthread.h>


void *print_message(void *vargp){
	char* message = (char*)arg;
	printf("%s\n", message);
	return NULL;
	
}

int main(){
	pthread_t thread1, thread2;
	
	//create first thread
	pthread_create(&thread1, NULL, print_message, "hello from thread 1");
	//create second thread
	pthread_create(&thread2, NULL, print_message, "hello from thread 2");
	
	//wait for both threads to finish
	pthread_join(thread1, NULL);
	pthread_join(thread2, NULL);
	
	printf("Both threads completed\n");
	return 0;
}

output:

[2021ict37@fedora ~]$ ./thread2
hello from thread 1
hello from thread 2
Both threads completed

//3.
#include<stdio.h>
#include<stdlib.h>
#include<unistd.h>
#include<pthread.h>


void *print_message(void *arg){
	char* message = (char*)arg;
	printf("%s\n", message);
	return NULL;
	
}

int main() {
	pthread_t threads[3];
	
	char* message[] = {
	"Thread1 says hi",
	"Thread2 says hello",
	"Thread3 says hey"
	};
	
	for (int i = 0; i < 3; i++){
	pthread_create(&threads[i], NULL, print_message, message[i]);
	}
	
	for (int i = 0; i < 3; i++){
	pthread_join(threads[i], NULL);
	}
	
	printf("All 3 threds done\n");
	return 0;
}

Output:

[2021ict37@fedora ~]$ ./thread4
Thread1 says hi
Thread2 says hello
Thread3 says hey
All 3 threds done
